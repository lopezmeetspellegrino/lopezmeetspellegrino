<script>
function go(n){
  document.querySelectorAll('.page').forEach(function(p){p.classList.remove('active')});
  document.querySelectorAll('nav a').forEach(function(a){a.classList.remove('active')});
  document.getElementById('page-'+n).classList.add('active');
  document.getElementById('nav-'+n).classList.add('active');
  window.scrollTo({top:0,behavior:'smooth'});
}
function tick(){
  var t=new Date('May 1, 2027 18:00:00').getTime()-Date.now();
  if(t<=0){['days','hours','mins','secs'].forEach(function(u){document.getElementById('cd-'+u).textContent='0'});return;}
  document.getElementById('cd-days').textContent=Math.floor(t/86400000);
  document.getElementById('cd-hours').textContent=Math.floor(t%86400000/3600000);
  document.getElementById('cd-mins').textContent=String(Math.floor(t%3600000/60000)).padStart(2,'0');
  document.getElementById('cd-secs').textContent=String(Math.floor(t%60000/1000)).padStart(2,'0');
}
tick();setInterval(tick,1000);
function submitRSVP(){
  document.getElementById('vform').style.display='none';
  document.getElementById('vok').style.display='block';
}
